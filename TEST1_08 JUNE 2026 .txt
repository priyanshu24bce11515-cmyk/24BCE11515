1. Display all CS students who scored more than 80 marks.

db.students.find({
dept: "CS",
marks: { $gt: 80 }
});

---

2. Retrieve students whose marks lie between 60 and 85 (inclusive).

db.students.find({
marks: {
$gte: 60,
$lte: 85
}
});

---

3. Find students belonging to either the IT or ECE department.

db.students.find({
dept: { $in: ["IT", "ECE"] }
});

---

4. Show students who are not from the CS department.

db.students.find(
{ dept: { $nin: ["CS"] } },
{ name: 1, dept: 1, _id: 0 }
);

---

5. Retrieve students whose city is Chennai.

db.students.find({
"address.city": "Chennai"
});

---

6. Calculate total revenue generated by each product category.

db.products.aggregate([
{
$group: {
_id: "$category",
totalRevenue: {
$sum: {
$multiply: ["$price", "$sold"]
}
}
}
}
]);

---

7. Find the total number of students and average marks department-wise.

db.students.aggregate([
{
$group: {
_id: "$dept",
totalStudents: { $sum: 1 },
averageMarks: { $avg: "$marks" }
}
}
]);

---

8. Remove the scholarship field from every student document.

db.students.updateMany(
{},
{
$unset: {
scholarship: ""
}
}
);

---

9. Add scholarship = false for all CS students.

db.students.updateMany(
{ dept: "CS" },
{
$set: {
scholarship: false
}
}
);

---

10. Increase marks of the student with _id = 6 by 10.

db.students.updateOne(
{ _id: 6 },
{
$inc: {
marks: 10
}
}
);

---

11. Retrieve the 3rd and 4th highest scorers.

db.students.find()
.sort({ marks: -1 })
.skip(2)
.limit(2);

---

12. Find employees having salary above 50,000 and experience greater than 5 years.

db.employees.find({
salary: { $gt: 50000 },
experience: { $gt: 5 }
});

---

13. Display employees whose skills array contains more than two elements.

db.employees.find({
$expr: {
$gt: [
{ $size: "$skills" },
2
]
}
});

---

14. Find departments where the average salary exceeds 50,000.

db.employees.aggregate([
{
$group: {
_id: "$department",
averageSalary: { $avg: "$salary" }
}
},
{
$match: {
averageSalary: { $gt: 50000 }
}
}
]);

---

15. Determine the minimum salary in each department.

db.employees.aggregate([
{
$group: {
_id: "$department",
minSalary: { $min: "$salary" }
}
}
]);

---

16. Rename the field "city" to "location".

db.employees.updateMany(
{},
{
$rename: {
city: "location"
}
}
);

---

17. Find products where the stock field is stored as a numeric data type.

db.products.find({
stock: { $type: "number" }
});

---

18. Change the active status to true for all inactive students.

db.students.updateMany(
{ active: false },
{
$set: {
active: true
}
}
);
